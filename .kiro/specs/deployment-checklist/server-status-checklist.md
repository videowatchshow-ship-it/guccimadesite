## 📝 검증 방법

각 항목은 다음과 같이 검증합니다:

```bash
#!/bin/bash

# 정규식 검증 함수
validate_regex() {
    local value=$1
    local regex=$2
    local name=$3
    
    if [[ $value =~ $regex ]]; then
        echo "✅ $name: $value (검증 통과)"
        return 0
    else
        echo "❌ $name: $value (검증 실패)"
        return 1
    fi
}

# 예제
validate_regex "76.13.218.129" "^[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}$" "IP 주소"

validate_regex "4Gi" "^[0-9]+Gi$" "메모리"
```
