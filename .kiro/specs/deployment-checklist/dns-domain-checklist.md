### 정규식 검증 규칙
```regex
# 도메인명
^[a-z0-9]([a-z0-9-]{0,61}[a-z0-9])?(\.[a-z0-9]([a-z0-9-]{0,61}[a-z0-9])?)*$

# IP 주소
^[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}$

# 네임서버
^ns[0-9]+\.[a-z0-9.-]+$

# TTL
^[0-9]{1,5}$
```

## 📊 검증 스크립트

```bash
#!/bin/bash

# 정규식 검증 함수
validate_regex() {
    local value=$1
    local regex=$2
    local name=$3
    
    if [[ $value =~ $regex ]]; then
        echo "✅ $name: $value (검증 통과)"
        return 0
    else
        echo "❌ $name: $value (검증 실패)"
        return 1
    fi
}

# DNS 검증 함수
validate_dns() {
    local domain=$1
    local expected_ip=$2
    
    local result=$(nslookup $domain | grep "Address:" | tail -1 | awk '{print $2}')
    
    if [[ $result == $expected_ip ]]; then
        echo "✅ DNS 조회: $domain -> $result (검증 통과)"
        return 0
    else
        echo "❌ DNS 조회: $domain -> $result (예상: $expected_ip)"
        return 1
    fi
}

# 예제
validate_regex "xn--2e0bj1fruw33b6ti.net" "^[a-z0-9]([a-z0-9-]{0,61}[a-z0-9])?(\.[a-z0-9]([a-z0-9-]{0,61}[a-z0-9])?)*$" "도메인명"
validate_regex "76.13.218.129" "^[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}$" "IP 주소"
validate_dns "xn--2e0bj1fruw33b6ti.net" "76.13.218.129"
```
